        $payment

                'post_type'=>
                    'ax_payment',


                'post_title'=>
                    'Payment ' .
                    time(),


                'post_status'=>
                    'publish'


            ]

        );


    }


}

/**
 * Athletix Invoice Manager
 */
class Athletix_Invoice_Manager {


    public static function init() {


        add_action(

            'init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_post_type(

            'ax_invoice',

            [

                'labels'=>[

                    'name'=>
                        'Invoices',

                    'singular_name'=>
                        'Invoice'

                ],


                'public'=>
                    false,


                'show_ui'=>
                    true,


                'supports'=>[

                    'title'

                ]

            ]

        );


    }





    public static function create(
        $user_id,
        $amount
    ) {


        $invoice_id =
            wp_insert_post(

                [

                    'post_type'=>
                        'ax_invoice',


                    'post_title'=>
                        'Invoice - ' .
                        $user_id,


                    'post_status'=>
                        'publish'

                ]

            );



        update_post_meta(

            $invoice_id,

            '_athletix_amount',

            floatval(

                $amount

            )

        );



        update_post_meta(

            $invoice_id,

            '_athletix_user',

            absint(

                $user_id

            )

        );



        return $invoice_id;


    }


}

/**
 * Athletix Subscriptions
 */
class Athletix_Subscriptions {


    public static function init() {


    }





    public static function create(
        $user_id,
        $plan_id,
        $interval = 'monthly'
    ) {


        update_user_meta(

            $user_id,

            '_athletix_subscription',

            [

                'plan'=>
                    absint(

                        $plan_id

                    ),


                'interval'=>
                    sanitize_key(

                        $interval

                    ),


                'active'=>
                    true

            ]

        );


    }





    public static function cancel(
        $user_id
    ) {


        delete_user_meta(

            $user_id,

            '_athletix_subscription'

        );


    }


}
