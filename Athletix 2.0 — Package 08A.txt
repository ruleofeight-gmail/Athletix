/**
 * Team Profile Display
 */
class Athletix_Team_Profile {


    public static function render(
        $atts
    ) {


        $atts =
            shortcode_atts(

                [

                    'id'=>0

                ],

                $atts

            );



        $team =
            get_post(

                absint(

                    $atts['id']

                )

            );



        if (
            ! $team
        ) {

            return '';

        }



        $repo =
            Athletix_Registry::get(

                'team.repository'

            );



        $stats =
            $repo->standings(

                $team->ID

            );



        ob_start();


        ?>


        <div class="athletix-team-profile">


            <h2>

                <?php echo esc_html(
                    $team->post_title
                ); ?>

            </h2>


            <div>

                Played:
                <?php echo esc_html(
                    $stats['played'] ?? 0
                ); ?>

            </div>


            <div>

                Points:
                <?php echo esc_html(
                    $stats['points'] ?? 0
                ); ?>

            </div>


        </div>


        <?php


        return ob_get_clean();


    }


}
