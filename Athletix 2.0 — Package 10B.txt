/**
 * Athletix Event Scheduler
 *
 * WordPress cron integration.
 */
class Athletix_Event_Scheduler {


    public static function init() {


        add_action(

            'athletix_daily_event_check',

            [

                __CLASS__,

                'run'

            ]

        );


        if (
            ! wp_next_scheduled(
                'athletix_daily_event_check'
            )
        ) {


            wp_schedule_event(

                time(),

                'daily',

                'athletix_daily_event_check'

            );


        }


    }





    public static function run() {


        Athletix_Reminder_Engine::process();


    }


}

        }

            $data['email'],

            $data['subject']

/**
 * Athletix Reminder Engine
 */
class Athletix_Reminder_Engine {


    public static function init() {


    }





    public static function process() {


        $matches =
            get_posts(

                [

                    'post_type'=>
                        'ax_match',

                    'posts_per_page'=>
                        -1,

                    'meta_query'=>[

                        [

                            'key'=>
                                '_athletix_match_status',

                            'value'=>
                                'scheduled'

                        ]

                    ]

                ]

            );



        foreach(
            $matches as $match
        ) {


            do_action(

                'athletix_match_reminder',

                $match->ID

            );


        }


    }


}

/**
 * Athletix Automation Rules
 */
class Athletix_Automation_Rules {


    public static function init() {


        add_action(

            'athletix_match_reminder',

            [

                __CLASS__,

                'match_reminder'

            ]

        );


    }





    public static function match_reminder(
        $match_id
    ) {


        Athletix_Notification_Engine::send(

            'email',

            [

                'email'=>
                    get_option(
                        'admin_email'
                    ),


                'subject'=>
                    'Upcoming Athletix Match',


                'message'=>
                    'Match #' .
                    $match_id .
                    ' is scheduled.'

            ]

        );


    }


}
