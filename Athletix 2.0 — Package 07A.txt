/**
 * Athletix Team REST Controller
 */
class Athletix_REST_Teams {


    public static function init() {


        add_action(

            'athletix_register_rest_routes',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_rest_route(

            'athletix/v1',

            '/teams',

            [

                'methods' =>
                    'GET',


                'callback' =>
                    [

                        __CLASS__,

                        'index'

                    ],

            ]

        );


    }





    public static function index() {


        $repository =
            Athletix_Registry::get(

                'team.repository'

            );



        $teams =
            $repository->all();



        $results = [];



        foreach (
            $teams as $team
        ) {


            $results[] = [

                'id' =>
                    $team->ID,


                'name' =>
                    $team->post_title,


                'standings' =>
                    $repository->standings(

                        $team->ID

                    )

            ];


        }



        return $results;


    }


}
