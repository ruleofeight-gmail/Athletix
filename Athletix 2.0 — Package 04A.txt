/**
 * Athletix Base Repository
 *
 * Common database operations.
 */
abstract class Athletix_Base_Repository {


    /**
     * Post type.
     */
    protected $post_type = '';



    /**
     * Find by ID.
     */
    public function find(
        $id
    ) {


        return get_post(
            absint($id)
        );


    }





    /**
     * Get all records.
     */
    public function all(
        $args = []
    ) {


        $defaults = [

            'post_type' =>
                $this->post_type,

            'posts_per_page' =>
                -1,

            'post_status' =>
                'publish'

        ];



        return get_posts(
            array_merge(
                $defaults,
                $args
            )
        );


    }





    /**
     * Create post.
     */
    public function create(
        $data
    ) {


        return wp_insert_post(
            $data
        );


    }





    /**
     * Update meta.
     */
    public function meta(
        $id,
        $key,
        $value
    ) {


        update_post_meta(
            $id,
            $key,
            $value
        );


    }





    /**
     * Get meta.
     */
    public function get_meta(
        $id,
        $key,
        $default = ''
    ) {


        $value =
            get_post_meta(
                $id,
                $key,
                true
            );


        return
            $value !== ''
            ?
            $value
            :
            $default;


    }


}

/**
 * Team Repository
 */
class Athletix_Team_Repository
extends Athletix_Base_Repository {


    protected $post_type =
        'ax_team';





    public function players(
        $team
    ) {


        return Athletix_Relationship_Engine::get(

            $team,

            'team_player'

        );


    }





    public function standings(
        $team
    ) {


        return $this->get_meta(

            $team,

            '_athletix_standings',

            []

        );


    }


}
