/**
 * Athletix Team Branding
 */
class Athletix_Team_Branding {


    public static function init() {


        add_shortcode(

            'athletix_team_branding',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render(
        $atts
    ) {


        $atts =
            shortcode_atts(

                [

                    'id'=>0

                ],

                $atts

            );



        $team_id =
            absint(
                $atts['id']
            );



        if(
            ! $team_id
        ) {

            return '';

        }



        $logo =
            Athletix_Logo_Manager::get(

                $team_id

            );



        ob_start();

        ?>

        <div class="athletix-team-branding">


            <?php if(
                $logo
            ) : ?>


                <?php echo Athletix_Image_Handler::html(

                    $logo,

                    'medium'

                ); ?>


            <?php endif; ?>


            <h2>

                <?php echo esc_html(

                    get_the_title(
                        $team_id
                    )

                ); ?>


            </h2>


        </div>

        <?php


        return ob_get_clean();


    }


}

/**
 * Athletix Media Gallery
 */
class Athletix_Media_Gallery {


    public static function init() {


        add_shortcode(

            'athletix_gallery',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render(
        $atts
    ) {


        $atts =
            shortcode_atts(

                [

                    'id'=>0

                ],

                $atts

            );



        $images =
            Athletix_Gallery::get(

                absint(

                    $atts['id']

                )

            );



        if(
            empty(
                $images
            )
        ) {

            return '';

        }



        ob_start();


        ?>


        <div class="athletix-gallery">


        <?php foreach(
            $images as $image
        ) : ?>


            <?php echo Athletix_Image_Handler::html(

                $image,

                'thumbnail'

            ); ?>


        <?php endforeach; ?>


        </div>


        <?php


        return ob_get_clean();


    }


}
