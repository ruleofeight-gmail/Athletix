/**
 * Athletix Calendar Engine
 */
class Athletix_Calendar_Engine {


    public static function init() {


    }





    public static function events(
        $args = []
    ) {


        return get_posts(

            [

                'post_type'=>
                    'ax_match',


                'posts_per_page'=>
                    $args['limit']
                    ??
                    -1,


                'meta_query'=>[

                    [

                        'key'=>
                            '_athletix_date',

                        'compare'=>
                            'EXISTS'

                    ]

                ]

            ]

        );


    }





    public static function add_event(
        $match_id,
        $date
    ) {


        update_post_meta(

            $match_id,

            '_athletix_date',

            sanitize_text_field(
                $date
            )

        );


    }


}

/**
 * Athletix Event Calendar
 */
class Athletix_Event_Calendar {


    public static function init() {


        add_shortcode(

            'athletix_calendar',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        $events =
            Athletix_Calendar_Engine::events();



        if(
            empty(
                $events
            )
        ) {

            return 'No scheduled events.';

        }



        ob_start();


        ?>


        <div class="athletix-calendar">


        <?php foreach(
            $events as $event
        ) : ?>


            <div class="calendar-event">


                <strong>

                    <?php echo esc_html(

                        $event->post_title

                    ); ?>


                </strong>


                <br>


                <?php echo esc_html(

                    get_post_meta(

                        $event->ID,

                        '_athletix_date',

                        true

                    )

                ); ?>


            </div>


        <?php endforeach; ?>


        </div>


        <?php


        return ob_get_clean();


    }


}

/**
 * Athletix Venue Manager
 */
class Athletix_Venue_Manager {


    public static function init() {


        add_action(

            'init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_post_type(

            'ax_venue',

            [

                'labels'=>[

                    'name'=>
                        'Venues',

                    'singular_name'=>
                        'Venue'

                ],


                'public'=>
                    false,


                'show_ui'=>
                    true,


                'supports'=>[

                    'title',

                    'editor'

                ]

            ]

        );


    }





    public static function assign(
        $match_id,
        $venue_id
    ) {


        update_post_meta(

            $match_id,

            '_athletix_venue',

            absint(

                $venue_id

            )

        );


    }


}

/**
 * Athletix Scheduling Conflicts
 */
class Athletix_Scheduling_Conflicts {


    public static function init() {


    }





    public static function check(
        $date,
        $venue_id
    ) {


        $matches =
            get_posts(

                [

                    'post_type'=>
                        'ax_match',


                    'meta_query'=>[

                        'relation'=>
                            'AND',


                        [

                            'key'=>
                                '_athletix_date',

                            'value'=>
                                $date

                        ],


                        [

                            'key'=>
                                '_athletix_venue',

                            'value'=>
                                $venue_id

                        ]

                    ]

                ]

            );



        return ! empty(
            $matches
        );


    }


}
