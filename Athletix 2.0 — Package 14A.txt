class Athletix_Search_Engine {


    public static function init() {


    }





    public static function search(
        $type,
        $query,
        $args = []
    ) {


        switch(
            $type
        ) {


            case 'team':

                return Athletix_Team_Search::find(

                    $query,

                    $args

                );

                return Athletix_Player_Search::find(

                    $query,

                    $args

                );


        }



        return [];


    }

/**
 * Athletix Filter Engine
 */
class Athletix_Filter_Engine {


    public static function init() {


    }





    public static function apply(
        $query,
        $filters = []
    ) {


        if(
            empty(
                $filters
            )
        ) {

            return $query;

        }



        foreach(
            $filters as $key=>$value
        ) {


            $query->set(

                'meta_query',

                [

                    [

                        'key'=>
                            sanitize_key(
                                $key
                            ),

                        'value'=>
                            sanitize_text_field(
                                $value
                            )

                    ]

                ]

            );


        }



        return $query;


    }


}
