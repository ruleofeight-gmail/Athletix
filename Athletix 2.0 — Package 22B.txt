/**
 * Athletix API Controller
 */
class Athletix_API_Controller {


    public static function init() {


        add_action(

            'rest_api_init',

            [

                __CLASS__,

                'routes'

            ]

        );


    }





    public static function routes() {


        register_rest_route(

            'athletix/v1',

            '/matches',

            [

                'methods'=>
                    'GET',


                'callback'=>

                    [

                        __CLASS__,

                        'matches'

                    ],


                'permission_callback'=>

                    [

                        'Athletix_API_Permissions',

                        'access'

                    ]

            ]

        );


    }





    public static function matches() {


        return get_posts(

            [

                'post_type'=>
                    'ax_match',


                'posts_per_page'=>
                    -1

            ]

        );


    }


}

/**
 * Athletix Mobile Endpoints
 */
class Athletix_Mobile_Endpoints {


    public static function init() {


        add_action(

            'rest_api_init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_rest_route(

            'athletix/v1',

            '/dashboard',

            [

                'methods'=>
                    'GET',


                'callback'=>

                    [

                        __CLASS__,

                        'dashboard'

                    ]

            ]

        );


    }





    public static function dashboard() {


        return [

            'teams'=>

                wp_count_posts(

                    'ax_team'

                )->publish,


            'players'=>

                wp_count_posts(

                    'ax_player'

                )->publish,


            'matches'=>

                wp_count_posts(

                    'ax_match'

                )->publish

        ];


    }


}
