/**
 * Athletix Membership Engine
 */
class Athletix_Membership_Engine {


    public static function init() {


    }





    public static function get_level(
        $user_id
    ) {


        $level =
            get_user_meta(

                $user_id,

                '_athletix_membership',

                true

            );



        if(
            empty(
                $level
            )
        ) {

            $level = 'standard';

        }



        return $level;


    }





    public static function assign(
        $user_id,
        $level
    ) {


        update_user_meta(

            $user_id,

            '_athletix_membership',

            sanitize_key(

                $level

            )

        );


    }


}

/**
 * Athletix Player Registration
 */
class Athletix_Player_Registration {


    public static function init() {


    }





    public static function register(
        $user_id,
        $player_data
    ) {


        $player_id =
            wp_insert_post(

                [

                    'post_type'=>
                        'ax_player',


                    'post_title'=>
                        sanitize_text_field(

                            $player_data['name']

                        ),


                    'post_status'=>
                        'publish'

                ]

            );



        if(
            $player_id
        ) {


            update_post_meta(

                $player_id,

                '_athletix_user',

                $user_id

            );


        }



        return $player_id;


    }


}

/**
 * Athletix Waiver Management
 */
class Athletix_Waivers {


    public static function init() {


    }





    public static function accept(
        $user_id,
        $waiver_id
    ) {


        $accepted =
            get_user_meta(

                $user_id,

                '_athletix_waivers',

                true

            );



        if(
            ! is_array(
                $accepted
            )
        ) {

            $accepted = [];

        }



        $accepted[] =
            absint(
                $waiver_id
            );



        update_user_meta(

            $user_id,

            '_athletix_waivers',

            array_unique(

                $accepted

            )

        );


    }





    public static function has_signed(
        $user_id,
        $waiver_id
    ) {


        $accepted =
            get_user_meta(

                $user_id,

                '_athletix_waivers',

                true

            );



        return in_array(

            $waiver_id,

            (array)$accepted

        );


    }


}
