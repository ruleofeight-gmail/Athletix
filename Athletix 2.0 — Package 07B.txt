/**
 * Athletix Match REST Controller
 */
class Athletix_REST_Matches {


    public static function init() {


        add_action(

            'athletix_register_rest_routes',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_rest_route(

            'athletix/v1',

            '/matches',

            [

                'methods' =>
                    'GET',


                'callback' =>
                    [

                        __CLASS__,

                        'index'

                    ]

            ]

        );


    }





    public static function index() {


        $repository =
            Athletix_Registry::get(

                'match.repository'

            );



        $matches =
            $repository->all();



        $results = [];



        foreach (
            $matches as $match
        ) {


            $results[] = [

                'id' =>
                    $match->ID,


                'title' =>
                    $match->post_title,


                'home_team' =>
                    $repository->home_team(

                        $match->ID

                    ),


                'away_team' =>
                    $repository->away_team(

                        $match->ID

                    ),


                'score' =>
                    $repository->score(

                        $match->ID

                    ),


                'season' =>
                    $repository->season(

                        $match->ID

                    )

            ];


        }



        return $results;


    }


}

/**
 * Athletix Standings REST Controller
 */
class Athletix_REST_Standings {


    public static function init() {


        add_action(

            'athletix_register_rest_routes',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_rest_route(

            'athletix/v1',

            '/standings/(?P<league>\d+)',

            [

                'methods' =>
                    'GET',


                'callback' =>
                    [

                        __CLASS__,

                        'show'

                    ]

            ]

        );


    }





    public static function show(
        $request
    ) {


        $league =
            absint(

                $request['league']

            );



        $repository =
            Athletix_Registry::get(

                'league.repository'

            );



        $teams =
            $repository->teams(

                $league

            );



        return Athletix_Ranking_Engine::rank(

            $teams

        );


    }


}
