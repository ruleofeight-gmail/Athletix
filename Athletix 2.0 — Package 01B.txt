/**
 * Athletix Plugin Core
 *
 * Main application controller.
 */
class Athletix_Plugin {


    /**
     * Boot application.
     */
    public static function boot() {


        /*
         * Load core services
         */
        self::load_core();



        /*
         * Fire framework event
         */
        do_action(
            'athletix_loaded'
        );


    }





    /**
     * Load framework services.
     */
    private static function load_core() {


        $services = [

            'Athletix_Hooks',
            'Athletix_Events',
            'Athletix_Config',
            'Athletix_Registry',
            'Athletix_Cache',
            'Athletix_Logger',
            'Athletix_Validator',

        ];



        foreach (
            $services as $service
        ) {


            if (
                class_exists(
                    $service
                )
            ) {


                if (
                    method_exists(
                        $service,
                        'init'
                    )
                ) {

                    $service::init();

                }


            }


        }


    }



}

/**
 * Athletix Activation Handler
 */
class Athletix_Activator {


    /**
     * Run activation tasks.
     */
    public static function activate() {


        /*
         * Store plugin version
         */
        update_option(
            'athletix_version',
            ATHLETIX_VERSION
        );



        /*
         * Framework status
         */
        update_option(
            'athletix_installed',
            time()
        );



        /*
         * Prepare rewrite rules
         */
        flush_rewrite_rules();


    }


}
