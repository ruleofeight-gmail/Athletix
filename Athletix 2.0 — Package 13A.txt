/**
 * Athletix Media Manager
 */
class Athletix_Media_Manager {


    public static function init() {


        add_action(

            'init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_post_type(

            'ax_media',

            [

                'labels'=>[

                    'name'=>
                        'Athletix Media',

                    'singular_name'=>
                        'Media'

                ],


                'public'=>
                    false,


                'show_ui'=>
                    true,


                'supports'=>[

                    'title',

                    'thumbnail'

                ]

            ]

        );


    }





    public static function attach(
        $object_id,
        $media_id
    ) {


        update_post_meta(

            $object_id,

            '_athletix_media',

            absint(
                $media_id
            )

        );


    }





    public static function get(
        $object_id
    ) {


        return get_post_meta(

            $object_id,

            '_athletix_media',

            true

        );


    }


}

/**
 * Athletix Image Handler
 */
class Athletix_Image_Handler {


    public static function init() {


    }





    public static function url(
        $attachment_id,
        $size = 'medium'
    ) {


        return wp_get_attachment_image_url(

            $attachment_id,

            $size

        );


    }





    public static function html(
        $attachment_id,
        $size = 'medium'
    ) {


        return wp_get_attachment_image(

            $attachment_id,

            $size

        );


    }


}

    }





    public static function add(
        $object_id,
        $images
    ) {


        update_post_meta(

            $object_id,

            '_athletix_gallery',

            array_map(

                'absint',

                $images

            )

        );

        $object_id
    ) {


        return get_post_meta(

            $object_id,
