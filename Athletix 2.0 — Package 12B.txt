/**
 * Athletix Access Control
 */
class Athletix_Access_Control {


    public static function init() {


        add_filter(

            'map_meta_cap',

            [

                __CLASS__,

                'filter'

            ],

            10,

            4

        );


    }





    public static function filter(
        $caps,
        $cap,
        $user_id,
        $args
    ) {


        if (
            $cap !== 'manage_athletix'
        ) {

            return $caps;

        }



        $user =
            get_userdata(
                $user_id
            );



        if(
            ! $user
        ) {

            return [

                'do_not_allow'

            ];

        }



        if(
            in_array(

                'athletix_manager',

                $user->roles

            )
        ) {


            return [

                'exist'

            ];


        }



        return [

            'do_not_allow'

        ];


    }


}

/**
 * Athletix Audit Log
 */
class Athletix_Audit_Log {


    public static function init() {


        add_action(

            'save_post',

            [

                __CLASS__,

                'record'

            ],

            20,

            3

        );


    }





    public static function record(
        $post_id,
        $post,
        $update
    ) {


        if(
            wp_is_post_revision(
                $post_id
            )
        ) {

            return;

        }



        if(
            strpos(

                $post->post_type,

                'ax_'

            ) !== 0

        ) {

            return;

        }



        self::write(

            [

                'user'=>

                    get_current_user_id(),


                'action'=>

                    $update
                    ?
                    'updated'
                    :
                    'created',


                'object'=>

                    $post->post_type,


                'id'=>

                    $post_id


            ]

        );


    }





    public static function write(
        $data
    ) {


        $logs =
            get_option(

                'athletix_audit_logs',

                []

            );



        $logs[] = [

            'time'=>
                current_time(
                    'mysql'
                ),

            'data'=>
                $data

        ];



        update_option(

            'athletix_audit_logs',

            $logs

        );


    }





    public static function all() {


        return get_option(

            'athletix_audit_logs',

            []

        );


    }


}
