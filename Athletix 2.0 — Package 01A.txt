<?php

if ( ! defined( 'ABSPATH' ) ) {
    exit;
}


/**
 * Athletix Autoloader
 *
 * Loads Athletix classes automatically.
 */
class Athletix_Loader {


    /**
     * Initialize loader.
     */
    public static function init() {


        spl_autoload_register(
            [
                __CLASS__,
                'autoload'
            ]
        );


    }



    /**
     * Load classes.
     *
     * Example:
     *
     * Athletix_Config
     *
     * becomes:
     *
     * includes/core/class-config.php
     */
    public static function autoload(
        $class
    ) {


        if (
            strpos(
                $class,
                'Athletix_'
            ) !== 0
        ) {

            return;

        }


        $class_name =
            str_replace(
                'Athletix_',
                '',
                $class
            );


        $parts =
            explode(
                '_',
                $class_name
            );


        $filename =
            'class-' .
            strtolower(
                implode(
                    '-',
                    $parts
                )
            ) .
            '.php';



        /*
         * Search locations
         */

        $locations = [

            ATHLETIX_PATH . 'includes/',
            ATHLETIX_PATH . 'includes/core/',
            ATHLETIX_PATH . 'includes/admin/',
            ATHLETIX_PATH . 'includes/api/',
            ATHLETIX_PATH . 'includes/engine/',
            ATHLETIX_PATH . 'includes/repositories/',
            ATHLETIX_PATH . 'includes/services/',

        ];



        foreach (
            $locations as $location
        ) {


            $file =
                $location .
                $filename;



            if (
                file_exists(
                    $file
                )
            ) {

                require_once $file;

                return;

            }


        }


    }


}
