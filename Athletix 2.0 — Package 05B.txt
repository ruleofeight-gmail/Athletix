/**
 * Athletix Admin Assets
 */
class Athletix_Admin_Assets {


    public static function init() {


        add_action(
            'admin_enqueue_scripts',
            [
                __CLASS__,
                'enqueue'
            ]
        );


    }





    public static function enqueue(
        $hook
    ) {


        if (
            strpos(
                $hook,
                'athletix'
            ) === false
        ) {

            return;

        }



        wp_enqueue_style(

            'athletix-admin',

            ATHLETIX_URL .
            'assets/css/admin.css',

            [],

            ATHLETIX_VERSION

        );



        wp_enqueue_script(

            'athletix-admin',

            ATHLETIX_URL .
            'assets/js/admin.js',

            [

                'jquery'

            ],

            ATHLETIX_VERSION,

            true

        );



        wp_localize_script(

            'athletix-admin',

            'AthletixAdmin',

            [

                'ajax_url' =>
                    admin_url(
                        'admin-ajax.php'
                    ),


                'nonce' =>
                    wp_create_nonce(
                        'athletix_admin'
                    )

            ]

        );


    }


}

/**
 * Athletix Admin Notices
 */
class Athletix_Admin_Notices {


    public static function init() {


        add_action(

            'admin_notices',

            [

                __CLASS__,

                'display'

            ]

        );


    }





    public static function display() {


        if (
            ! current_user_can(
                'manage_options'
            )
        ) {

            return;

        }



        $installed =
            get_option(
                'athletix_installed'
            );



        if (
            empty($installed)
        ) {


            ?>

            <div class="notice notice-info">

                <p>

                Athletix setup is incomplete.

                Please configure your sport settings.

                </p>

            </div>

            <?php


        }


    }


}
