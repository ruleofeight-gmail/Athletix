/**
 * Match Meta Box
 *
 * Handles:
 * - home team
 * - away team
 * - scores
 * - match status
 */
class Athletix_Match_Meta_Box {


    public static function init() {


        add_action(

            'athletix_register_meta_boxes',

            [

                __CLASS__,

                'register'

            ]

        );


        add_action(

            'save_post_ax_match',

            [

                __CLASS__,

                'save'

            ]

        );


    }





    public static function register() {


        add_meta_box(

            'athletix_match_details',

            'Match Details',

            [

                __CLASS__,

                'render'

            ],

            'ax_match',

            'normal',

            'high'

        );


    }





    public static function render(
        $post
    ) {


        $home =
            get_post_meta(

                $post->ID,

                '_athletix_home_team',

                true

            );


        $away =
            get_post_meta(

                $post->ID,

                '_athletix_away_team',

                true

            );


        $scores =
            get_post_meta(

                $post->ID,

                '_athletix_scores',

                true

            );



        $home_score =
            $scores['home']
            ??
            0;



        $away_score =
            $scores['away']
            ??
            0;



        ?>


        <p>

            <label>
                Home Team ID
            </label>

            <input
                type="number"
                name="athletix_home_team"
                value="<?php echo esc_attr($home); ?>"
                class="widefat"
            >

        </p>



        <p>

            <label>
                Away Team ID
            </label>

            <input
                type="number"
                name="athletix_away_team"
                value="<?php echo esc_attr($away); ?>"
                class="widefat"
            >

        </p>



        <p>

            <label>
                Home Score
            </label>


            <input
                type="number"
                name="athletix_home_score"
                value="<?php echo esc_attr($home_score); ?>"
            >

        </p>




        <p>

            <label>
                Away Score
            </label>


            <input
                type="number"
                name="athletix_away_score"
                value="<?php echo esc_attr($away_score); ?>"
            >

        </p>


        <?php


    }





    public static function save(
        $post_id
    ) {


        if (
            isset(
                $_POST['athletix_home_team']
            )
        ) {


            update_post_meta(

                $post_id,

                '_athletix_home_team',

                absint(

                    $_POST['athletix_home_team']

                )

            );


        }



        if (
            isset(
                $_POST['athletix_away_team']
            )
        ) {


            update_post_meta(

                $post_id,

                '_athletix_away_team',

                absint(

                    $_POST['athletix_away_team']

                )

            );


        }



        if (

            isset(

                $_POST['athletix_home_score'],

                $_POST['athletix_away_score']

            )

        ) {


            update_post_meta(

                $post_id,

                '_athletix_scores',

                [

                    'home' =>
                        absint(

                            $_POST['athletix_home_score']

                        ),

                    'away' =>
                        absint(

                            $_POST['athletix_away_score']

                        )

                ]

            );


        }


    }
