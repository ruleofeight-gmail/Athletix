/**
 * Dashboard Upcoming Matches
 */
class Athletix_Dashboard_Matches {


    public static function init() {


        add_action(

            'wp_dashboard_setup',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        wp_add_dashboard_widget(

            'athletix_upcoming_matches',

            'Upcoming Matches',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        $matches =
            get_posts(

                [

                    'post_type'=>
                        'ax_match',

                    'posts_per_page'=>
                        5,

                    'post_status'=>
                        'publish'

                ]

            );



        if(
            empty(
                $matches
            )
        ) {

            echo 'No upcoming matches.';

            return;

        }



        echo '<ul>';



        foreach(
            $matches as $match
        ) {


            echo '<li>';

            echo esc_html(

                $match->post_title

            );

            echo '</li>';


        }



        echo '</ul>';


    }


}

/**
 * Dashboard Activity Feed
 */
class Athletix_Dashboard_Activity {


    public static function init() {


        add_action(

            'wp_dashboard_setup',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        wp_add_dashboard_widget(

            'athletix_activity',

            'Recent Activity',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        $logs =
            array_reverse(

                Athletix_Audit_Log::all()

            );



        $logs =
            array_slice(

                $logs,

                0,

                5

            );



        if(
            empty(
                $logs
            )
        ) {

            echo 'No recent activity.';

            return;

        }



        echo '<ul>';



        foreach(
            $logs as $log
        ) {


            echo '<li>';

            echo esc_html(

                $log['time']

            );

            echo ' - ';

            echo esc_html(

                $log['data']['action']

            );

            echo '</li>';


        }



        echo '</ul>';


    }


}
