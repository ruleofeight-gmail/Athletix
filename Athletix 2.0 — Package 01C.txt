/**
 * Athletix Logger
 *
 * Lightweight internal logging system.
 *
 * Stores framework events for debugging
 * and diagnostics.
 */
class Athletix_Logger {


    /**
     * Initialize logger.
     */
    public static function init() {

    }



    /**
     * Add log entry.
     *
     * @param string $level
     * @param string $message
     * @param array  $context
     */
    public static function log(
        $level,
        $message,
        $context = []
    ) {


        $logs = get_option(
            'athletix_logs',
            []
        );


        $logs[] = [

            'time' =>
                current_time(
                    'mysql'
                ),

            'level' =>
                sanitize_text_field(
                    $level
                ),

            'message' =>
                sanitize_text_field(
                    $message
                ),

            'context' =>
                $context

        ];



        /*
         * Keep log size controlled.
         */
        if (
            count($logs) > 250
        ) {

            $logs =
                array_slice(
                    $logs,
                    -250
                );

        }



        update_option(
            'athletix_logs',
            $logs
        );


    }



    /**
     * Information log.
     */
    public static function info(
        $message,
        $context = []
    ) {


        self::log(
            'info',
            $message,
            $context
        );


    }



    /**
     * Error log.
     */
    public static function error(
        $message,
        $context = []
    ) {


        self::log(
            'error',
            $message,
            $context
        );


    }



}

/**
 * Athletix Validator
 *
 * Common validation utilities.
 */
class Athletix_Validator {


    /**
     * Initialize.
     */
    public static function init() {

    }



    /**
     * Check required value.
     */
    public static function required(
        $value
    ) {


        return ! empty(
            $value
        );


    }



    /**
     * Validate post ID.
     */
    public static function post_id(
        $id
    ) {


        return absint(
            $id
        ) > 0;


    }



    /**
     * Validate array.
     */
    public static function array(
        $value
    ) {


        return is_array(
            $value
        );


    }



    /**
     * Sanitize text.
     */
    public static function text(
        $value
    ) {


        return sanitize_text_field(
            $value
        );


    }



}
