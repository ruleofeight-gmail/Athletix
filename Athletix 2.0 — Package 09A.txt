/**
 * Athletix CSV Importer
 */
class Athletix_CSV_Importer {


    /**
     * Import teams.
     */
    public static function teams(
        $file
    ) {


        $rows =
            self::read(
                $file
            );



        $count = 0;



        foreach(
            $rows as $row
        ) {


            wp_insert_post(

                [

                    'post_type'=>
                        'ax_team',

                    'post_title'=>
                        sanitize_text_field(

                            $row['name']

                        ),

                    'post_status'=>
                        'publish'

                ]

            );


            $count++;


        }



        return $count;


    }





    /**
     * Import players.
     */
    public static function players(
        $file
    ) {


        $rows =
            self::read(
                $file
            );



        $count = 0;



        foreach(
            $rows as $row
        ) {


            wp_insert_post(

                [

                    'post_type'=>
                        'ax_player',

                    'post_title'=>
                        sanitize_text_field(

                            $row['name']

                        ),

                    'post_status'=>
                        'publish'

                ]

            );


            $count++;


        }



        return $count;


    }





    /**
     * Import matches.
     */
    public static function matches(
        $file
    ) {


        return count(

            self::read(
                $file
            )

        );


    }





    /**
     * Read CSV.
     */
    private static function read(
        $file
    ) {


        $rows = [];



        if(
            ! file_exists(
                $file
            )
        ) {

            return $rows;

        }



        $handle =
            fopen(
                $file,
                'r'
            );



        $headers =
            fgetcsv(
                $handle
            );



        while(
            $line =
                fgetcsv(
                    $handle
                )
        ) {


            $rows[] =
                array_combine(

                    $headers,

                    $line

                );


        }



        fclose(
            $handle
        );



        return $rows;


    }


}

/**
 * Athletix Backup Exporter
 */
class Athletix_Backup_Exporter {


    public static function export(
        $type
    ) {


        $posts =
            get_posts(

                [

                    'post_type'=>
                        $type,

                    'posts_per_page'=>
                        -1

                ]

            );



        $data = [];



        foreach(
            $posts as $post
        ) {


            $data[] = [

                'id'=>
                    $post->ID,

                'name'=>
                    $post->post_title

            ];


        }



        return $data;


    }


}
