/**
 * Athletix Tournament Builder
 */
class Athletix_Tournament_Builder {


    public static function init() {


    }





    public static function create(
        $name,
        $teams = []
    ) {


        $tournament_id =
            wp_insert_post(

                [

                    'post_type'=>
                        'ax_league',


                    'post_title'=>
                        sanitize_text_field(
                            $name
                        ),


                    'post_status'=>
                        'publish'

                ]

            );



        if(
            ! $tournament_id
        ) {

            return false;

        }



        update_post_meta(

            $tournament_id,

            '_athletix_teams',

            array_map(

                'absint',

                $teams

            )

        );



        return $tournament_id;


    }


}

/**
 * Athletix Schedule Generator
 */
class Athletix_Schedule_Generator {


    public static function init() {


    }





    public static function round_robin(
        $teams
    ) {


        $matches = [];



        foreach(
            $teams as $home
        ) {


            foreach(
                $teams as $away
            ) {


                if(
                    $home === $away
                ) {

                    continue;

                }



                $matches[] = [

                    'home'=>
                        $home,


                    'away'=>
                        $away

                ];


            }


        }



        return $matches;


    }





    public static function create_matches(
        $schedule
    ) {


        foreach(
            $schedule as $game
        ) {


            wp_insert_post(

                [

                    'post_type'=>
                        'ax_match',


                    'post_title'=>
                        'Scheduled Match',


                    'post_status'=>
                        'publish'

                ]

            );


        }


    }


}
