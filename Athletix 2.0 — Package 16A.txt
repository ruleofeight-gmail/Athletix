/**
 * Athletix Divisions
 */
class Athletix_Divisions {


    public static function init() {


        add_action(

            'init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_post_type(

            'ax_division',

            [

                'labels'=>[

                    'name'=>
                        'Divisions',

                    'singular_name'=>
                        'Division'

                ],


                'public'=>
                    false,


                'show_ui'=>
                    true,


                'supports'=>[

                    'title'

                ]

            ]

        );


    }





    public static function get(
        $league_id
    ) {


        return get_posts(

            [

                'post_type'=>
                    'ax_division',


                'meta_query'=>[

                    [

                        'key'=>
                            '_athletix_league',

                        'value'=>
                            $league_id

                    ]

                ]

            ]

        );


    }


}

/**
 * Athletix Tournament Brackets
 */
class Athletix_Brackets {


    public static function init() {


    }





    public static function create(
        $teams
    ) {


        $rounds = [];



        while(
            count(
                $teams
            ) > 1
        ) {


            $rounds[] =
                array_chunk(

                    $teams,

                    2

                );


            $teams =
                array_slice(

                    $teams,

                    0,

                    ceil(

                        count(
                            $teams
                        ) / 2

                    )

                );


        }



        return $rounds;


    }


}

    }





    public static function get(
        $league_id
    ) {


        return get_post_meta(

            $league_id,

            '_athletix_playoffs',

            true

        );


    }





    public static function save(
        $league_id,
        $data
    ) {


        update_post_meta(

            $league_id,

            '_athletix_playoffs',

            $data

        );


    }


}
