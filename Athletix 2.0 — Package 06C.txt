<?php

if ( ! defined( 'ABSPATH' ) ) {
    exit;

/**
 * Player Statistics Meta Box
 *
 * Stores match performance data.
 */
class Athletix_Player_Statistics_Meta_Box {


    public static function init() {


        add_action(

            'athletix_register_meta_boxes',

            [

                __CLASS__,

                'register'

            ]

        );


        add_action(

            'save_post_ax_match',

            [

                __CLASS__,

                'save'

            ]

        );


    }





    public static function register() {


        add_meta_box(

            'athletix_player_statistics',

            'Player Statistics',

            [

                __CLASS__,

                'render'

            ],

            'ax_match',

            'normal',

            'default'

        );


    }





    public static function render(
        $post
    ) {


        $stats =
            get_post_meta(

                $post->ID,

                '_athletix_statistics',

                true

            );



        if (
            ! is_array($stats)
        ) {

            $stats = [];

        }



        ?>


        <p>

        Enter player statistics
        as JSON.

        </p>



        <textarea

            name="athletix_statistics"

            style="width:100%;height:150px;"

        ><?php

            echo esc_textarea(

                wp_json_encode(

                    $stats,

                    JSON_PRETTY_PRINT

                )

            );

        ?></textarea>



        <?php


    }





    public static function save(
        $post_id
    ) {


        if (
            empty(
                $_POST['athletix_statistics']
            )
        ) {

            return;

        }



        $json =
            stripslashes(

                $_POST['athletix_statistics']

            );



        $data =
            json_decode(

                $json,

                true

            );



        if (
            is_array($data)
        ) {


            update_post_meta(

                $post_id,

                '_athletix_statistics',

                $data

            );


        }


    }


}
