<?php

    exit;
}


/**
 * Athletix Relationship Engine
 *
 * Central relationship storage.
 */
class Athletix_Relationship_Engine {


    public static function init() {

    }



    /**
     * Create relationship.
     */
    public static function connect(
        $parent,
        $child,
        $type
    ) {


        $relationships =
            get_post_meta(
                $parent,
                '_athletix_relationships',
                true
            );


        if (
            ! is_array(
                $relationships
            )
        ) {

            $relationships = [];

        }



        $relationships[] = [

            'child' =>
                absint(
                    $child
                ),

            'type' =>
                sanitize_key(
                    $type
                )

        ];



        update_post_meta(
            $parent,
            '_athletix_relationships',
            $relationships
        );


    }





    /**
     * Get related items.
     */
    public static function get(
        $parent,
        $type = ''
    ) {


        $relationships =
            get_post_meta(
                $parent,
                '_athletix_relationships',
                true
            );


        if (
            ! is_array(
                $relationships
            )
        ) {

            return [];

        }



        $results = [];



        foreach (
            $relationships as $relationship
        ) {


            if (
                $type === ''
                ||
                $relationship['type'] === $type
            ) {


                $results[] =
                    $relationship['child'];


            }


        }



        return $results;


    }



}

/**
 * League → Season Relationship
 */
class Athletix_League_Season {


    public static function init() {


        add_action(
            'athletix_register_relationships',
            [
                __CLASS__,
                'register'
            ]
        );


    }



    public static function register() {

    }



    public static function attach(
        $league,
        $season
    ) {


        Athletix_Relationship_Engine::connect(
            $league,
            $season,
            'league_season'
        );


    }


}

/**
 * Match Participants
 *
 * Connects matches to teams.
 */
class Athletix_Match_Participants {


    public static function init() {

    }



    public static function set_home_team(
        $match,
        $team
    ) {


        update_post_meta(
            $match,
            '_athletix_home_team',
            absint(
                $team
            )
        );


    }




    public static function set_away_team(
        $match,
        $team
    ) {


        update_post_meta(
            $match,
            '_athletix_away_team',
            absint(
                $team
            )
        );


    }




    public static function get_home_team(
        $match
    ) {


        return absint(
            get_post_meta(
                $match,
                '_athletix_home_team',
                true
            )
        );


    }



    public static function get_away_team(
        $match
    ) {


        return absint(
            get_post_meta(
                $match,
                '_athletix_away_team',
                true
            )
        );


    }


}
