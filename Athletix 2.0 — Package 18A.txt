/**
 * Athletix Notification Engine
 */
class Athletix_Notification_Engine {


    public static function init() {


    }





    public static function send(
        $type,
        $recipient,
        $message
    ) {


        switch(
            $type
        ) {


            case 'email':

                return Athletix_Email_Notifications::send(

                    $recipient,

                    $message

                );



            case 'alert':

                return Athletix_Alert_Manager::create(

                    $recipient,

                    $message

                );


        }



        return false;


    }





    public static function broadcast(
        $users,
        $message
    ) {


        foreach(
            $users as $user
        ) {


            self::send(

                'email',

                $user,

                $message

            );


        }


    }


}

/**
 * Athletix Email Notifications
 */
class Athletix_Email_Notifications {


    public static function init() {


    }





    public static function send(
        $email,
        $message
    ) {


        if(
            ! is_email(
                $email
            )
        ) {

            return false;

        }



        return wp_mail(

            $email,

            'Athletix Notification',

            wp_strip_all_tags(

                $message

            )

        );


    }


}

/**
 * Athletix Alert Manager
 */
class Athletix_Alert_Manager {


    public static function init() {


    }





    public static function create(
        $user_id,
        $message
    ) {


        $alerts =
            get_user_meta(

                $user_id,

                '_athletix_alerts',

                true

            );



        if(
            ! is_array(
                $alerts
            )
        ) {

            $alerts = [];

        }



        $alerts[] = [

            'message'=>
                sanitize_text_field(
                    $message
                ),


            'time'=>
                current_time(
                    'mysql'
                )

        ];



        update_user_meta(

            $user_id,

            '_athletix_alerts',

            $alerts

        );



        return true;


    }





    public static function get(
        $user_id
    ) {


        return get_user_meta(

            $user_id,

            '_athletix_alerts',

            true

        );


    }


}

/**
 * Athletix Message Center
 */
class Athletix_Message_Center {


    public static function init() {


        add_shortcode(

            'athletix_messages',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        if(
            ! is_user_logged_in()
        ) {

            return '';

        }



        $alerts =
            Athletix_Alert_Manager::get(

                get_current_user_id()

            );



        if(
            empty(
                $alerts
            )
        ) {

            return 'No messages.';

        }



        ob_start();


        ?>


        <div class="athletix-message-center">


        <?php foreach(
            $alerts as $alert
        ) : ?>


            <div class="message">


                <?php echo esc_html(

                    $alert['message']

                ); ?>


            </div>


        <?php endforeach; ?>


        </div>


        <?php


        return ob_get_clean();


    }


}
