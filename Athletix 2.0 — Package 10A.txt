/**
 * Athletix Report Engine
 *
 * Central reporting service.
 */
class Athletix_Report_Engine {


    public static function init() {


    }





    /**
     * Generate report.
     */
    public static function generate(
        $type,
        $args = []
    ) {


        switch(
            $type
        ) {


            case 'team':

                return Athletix_Team_Reports::generate(
                    $args
                );



            case 'player':

                return Athletix_Player_Reports::generate(
                    $args
                );



            case 'league':

                return Athletix_League_Reports::generate(
                    $args
                );


        }



        return [];

    }


}

        $league =

                ??
                0

            );



        if(
            ! $league
        ) {

            return [];

        }



        $repository =
            Athletix_Registry::get(

            );



        $teams =
            $repository->teams(

            );

            'league' =>
                get_the_title(
