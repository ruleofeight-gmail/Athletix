/**
 * Athletix Recurring Events
 */
class Athletix_Recurring_Events {


    public static function init() {


    }





    public static function generate(
        $start_date,
        $weeks,
        $day_interval = 7
    ) {


        $events = [];



        $date =
            strtotime(
                $start_date
            );



        for(
            $i = 0;
            $i < $weeks;
            $i++
        ) {


            $events[] =
                date(

                    'Y-m-d',

                    $date

                );



            $date =
                strtotime(

                    '+' .
                    $day_interval .
                    ' days',

                    $date

                );


        }



        return $events;


    }


}

/**
 * Athletix Season Calendar
 */
class Athletix_Season_Calendar {


    public static function init() {


    }





    public static function create(
        $season_id,
        $matches
    ) {


        foreach(
            $matches as $match
        ) {


            Athletix_Calendar_Engine::add_event(

                $match,

                current_time(
                    'Y-m-d'
                )

            );


        }



        update_post_meta(

            $season_id,

            '_athletix_calendar_created',

            true

        );


    }


}

/**
 * Athletix Public Schedule
 */
class Athletix_Public_Schedule {


    public static function init() {


        add_shortcode(

            'athletix_schedule',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        $events =
            Athletix_Calendar_Engine::events();



        ob_start();


        ?>


        <div class="athletix-schedule">


        <?php foreach(
            $events as $event
        ) : ?>


            <div class="schedule-item">


                <h4>

                    <?php echo esc_html(

                        $event->post_title

                    ); ?>


                </h4>


                <span>

                    <?php echo esc_html(

                        get_post_meta(

                            $event->ID,

                            '_athletix_date',

                            true

                        )

                    ); ?>


                </span>


            </div>


        <?php endforeach; ?>


        </div>


        <?php


        return ob_get_clean();


    }


}

/**
 * Athletix Calendar Export
 */
class Athletix_Calendar_Export {


    public static function init() {


        add_action(

            'init',

            [

                __CLASS__,

                'download'

            ]

        );


    }





    public static function download() {


        if(
            empty(
                $_GET['athletix_calendar']
            )
        ) {

            return;

        }



        $events =
            Athletix_Calendar_Engine::events();



        header(
            'Content-Type:text/calendar'
        );



        echo "BEGIN:VCALENDAR\n";



        foreach(
            $events as $event
        ) {


            echo "BEGIN:VEVENT\n";

            echo "SUMMARY:" .
                $event->post_title .
                "\n";

            echo "END:VEVENT\n";


        }



        echo "END:VCALENDAR";



        exit;


    }


}
