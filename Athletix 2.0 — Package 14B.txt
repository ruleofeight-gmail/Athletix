/**
 * Athletix Search Results
 */
class Athletix_Search_Results {


    public static function init() {


        add_shortcode(

            'athletix_search_results',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        if(
            empty(
                $_GET['athletix_query']
            )
        ) {

            return '';

        }



        $type =
            sanitize_key(

                $_GET['athletix_search_type']

            );



        $query =
            sanitize_text_field(

                $_GET['athletix_query']

            );



        $results =
            Athletix_Search_Engine::search(

                $type,

                $query

            );



        ob_start();


        ?>


        <div class="athletix-search-results">


        <?php foreach(
            $results as $item
        ) : ?>


            <div class="result">


                <a href="<?php echo esc_url(

                    get_permalink(
                        $item->ID
                    )

                ); ?>">


                    <?php echo esc_html(

                        $item->post_title

                    ); ?>


                </a>


            </div>


        <?php endforeach; ?>


        </div>


        <?php


        return ob_get_clean();


    }


}

/**
 * Athletix Directory
 */
class Athletix_Directory {


    public static function init() {


        add_shortcode(

            'athletix_directory',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render(
        $atts
    ) {


        $atts =
            shortcode_atts(

                [

                    'type'=>'player'

                ],

                $atts

            );



        $posts =
            get_posts(

                [

                    'post_type'=>

                        'ax_' .
                        sanitize_key(

                            $atts['type']

                        ),


                    'posts_per_page'=>
                        -1

                ]

            );



        ob_start();


        ?>


        <div class="athletix-directory">


        <?php foreach(
            $posts as $post
        ) : ?>


            <div class="directory-item">


                <a href="<?php echo esc_url(

                    get_permalink(
                        $post->ID
                    )

                ); ?>">


                    <?php echo esc_html(

                        $post->post_title

                    ); ?>


                </a>


            </div>


        <?php endforeach; ?>


        </div>


        <?php


        return ob_get_clean();


    }


}
