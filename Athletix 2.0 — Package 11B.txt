/**
 * Athletix Template Engine
 */
class Athletix_Template_Engine {


    public static function init() {


    }





    /**
     * Replace variables.
     */
    public static function render(
        $template,
        $data = []
    ) {


        foreach(
            $data as $key => $value
        ) {


            $template =
                str_replace(

                    '{{' . $key . '}}',

                    $value,

                    $template

                );


        }



        return $template;


    }


}

/**
 * Athletix Automation Rules Manager
 */
class Athletix_Rule_Manager {


    public static function init() {


    }





    public static function rules() {


        return get_option(

            'athletix_automation_rules',

            []

        );


    }





    public static function add(
        $rule
    ) {


        $rules =
            self::rules();



        $rules[] =
            $rule;



        update_option(

            'athletix_automation_rules',

            $rules

        );


    }





    public static function remove(
        $index
    ) {


        $rules =
            self::rules();



        unset(

            $rules[$index]

        );



        update_option(

            'athletix_automation_rules',

            array_values(
                $rules
            )

        );


    }


}

class Athletix_Admin_Automation {

        add_action(

            'admin_menu',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        add_submenu_page(

            'athletix',

            'Automation',

            'Automation',

            'manage_options',

            'athletix-automation',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        $rules =
            Athletix_Rule_Manager::rules();


        ?>


        <div class="wrap">


            <h1>
                Athletix Automation
            </h1>



            <p>

                Configure automated league actions.

            </p>



            <table class="widefat">


                <thead>

                    <tr>

                        <th>
                            Event

                            Action
                        </th>

                    </tr>

                </thead>



                <tbody>

                    $rules as $rule
