/**
 * Athletix Ranking Engine
 *
 * Sorts teams and generates league rankings.
 */
class Athletix_Ranking_Engine {


    public static function init() {

    }





    /**
     * Get ranked teams.
     *
     * @param array $teams
     *
     * @return array
     */
    public static function rank(
        $teams = []
    ) {


        $rankings = [];



        foreach (
            $teams as $team
        ) {


            $stats =
                get_post_meta(
                    $team,
                    '_athletix_standings',
                    true
                );



            if (
                ! is_array($stats)
            ) {

                continue;

            }



            $rankings[$team] =
                $stats;


        }



        uasort(
            $rankings,
            [

                __CLASS__,
                'sort'

            ]
        );



        return $rankings;


    }





    /**
     * Ranking comparison.
     */
    private static function sort(
        $a,
        $b
    ) {


        /*
         * Points first
         */
        if (
            $a['points']
            !==
            $b['points']
        ) {


            return
                $b['points']
                <=>
                $a['points'];


        }



        /*
         * Goal difference second
         */
        $a_diff =
            $a['goals_for']
            -
            $a['goals_against'];



        $b_diff =
            $b['goals_for']
            -
            $b['goals_against'];



        return
            $b_diff
            <=>
            $a_diff;


    }

/**
 * Athletix Schedule Engine
 *
 * Handles fixtures and calendars.
 */
class Athletix_Schedule_Engine {


    public static function init() {


    }





    /**
     * Create fixture.
     */
    public static function create(
        $season,
        $home,
        $away,
        $date
    ) {


        $match_id =
            wp_insert_post(

                [

                    'post_type' =>
                        'ax_match',

                    'post_title' =>
                        get_the_title($home)
                        .
                        ' vs '
                        .
                        get_the_title($away),

                    'post_status' =>
                        'publish'

                ]

            );



        if (
            $match_id
        ) {


            update_post_meta(

                $match_id,

                '_athletix_season',

                absint(
                    $season
                )

            );


            update_post_meta(

                $match_id,

                '_athletix_match_date',

                sanitize_text_field(
                    $date
                )

            );


            Athletix_Match_Participants::set_home_team(
                $match_id,
                $home
            );


            Athletix_Match_Participants::set_away_team(
                $match_id,
                $away
            );


        }



        return $match_id;


    }





    /**
     * Get season schedule.
     */
    public static function fixtures(
        $season
    ) {


        return get_posts(

            [

                'post_type' =>
                    'ax_match',

                'posts_per_page' =>
                    -1,

                'meta_query' => [

                    [

                        'key' =>
                            '_athletix_season',

                        'value' =>
                            absint($season)

                    ]

                ]

            ]

        );


    }



}
