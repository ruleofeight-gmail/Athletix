/**
 * Athletix Match Reminders
 */
class Athletix_Match_Reminders {


    public static function init() {


        add_action(

            'athletix_daily_tasks',

            [

                __CLASS__,

                'send'

            ]

        );


    }





    public static function send() {


        $matches =
            Athletix_Calendar_Engine::events();



        foreach(
            $matches as $match
        ) {


            $date =
                get_post_meta(

                    $match->ID,

                    '_athletix_date',

                    true

                );



            if(
                $date ===
                current_time(
                    'Y-m-d'
                )
            ) {


                self::notify(

                    $match

                );


            }


        }


    }





    private static function notify(
        $match
    ) {


        $message =
            'Your match is today: ' .
            $match->post_title;



        Athletix_Notification_Engine::broadcast(

            self::recipients(
                $match
            ),

            $message

        );


    }





    private static function recipients(
        $match
    ) {


        return [];


    }


}

/**
 * Athletix Score Alerts
 */
class Athletix_Score_Alerts {


    public static function init() {


        add_action(

            'athletix_score_updated',

            [

                __CLASS__,

                'send'

            ],

            10,

            2

        );


    }





    public static function send(
        $match_id,
        $score
    ) {


        $match =
            get_post(
                $match_id
            );



        if(
            ! $match
        ) {

            return;

        }



        Athletix_Notification_Engine::broadcast(

            [],

            'Score update: ' .
            $match->post_title .
            ' ' .
            $score

        );


    }


}

/**
 * Athletix Team Announcements
 */
class Athletix_Team_Announcements {


    public static function init() {


        add_action(

            'admin_post_athletix_announcement',

            [

                __CLASS__,

                'send'

            ]

        );


    }





    public static function send() {


        if(
            ! current_user_can(
                'manage_athletix'
            )
        ) {

            return;

        }



        $message =
            sanitize_text_field(

                $_POST['message']

            );



        Athletix_Notification_Engine::broadcast(

            [],

            $message

        );


        wp_safe_redirect(

            wp_get_referer()

        );


        exit;


    }


}

/**
 * Athletix Notification Preferences
 */
class Athletix_Notification_Preferences {


    public static function init() {


    }





    public static function get(
        $user_id
    ) {


        $settings =
            get_user_meta(

                $user_id,

                '_athletix_notifications',

                true

            );



        if(
            ! is_array(
                $settings
            )
        ) {


            $settings = [

                'email'=>true,

                'alerts'=>true,

                'scores'=>true

            ];


        }



        return $settings;


    }





    public static function save(
        $user_id,
        $settings
    ) {


        update_user_meta(

            $user_id,

            '_athletix_notifications',

            $settings

        );


    }


}
