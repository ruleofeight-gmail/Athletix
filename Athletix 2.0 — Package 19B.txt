/**
 * Athletix Analytics Dashboard
 */
class Athletix_Dashboard_Analytics {


    public static function init() {


        add_action(

            'wp_dashboard_setup',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        wp_add_dashboard_widget(

            'athletix_analytics',

            'Athletix Analytics',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        $teams =
            wp_count_posts(

                'ax_team'

            )->publish;



        $players =
            wp_count_posts(

                'ax_player'

            )->publish;



        ?>


        <div class="athletix-analytics">


            <p>

                Teams:
                <?php echo esc_html(
                    $teams
                ); ?>

            </p>


            <p>

                Players:
                <?php echo esc_html(
                    $players
                ); ?>

            </p>


        </div>


        <?php


    }


}

/**
 * Athletix Performance Trends
 */
class Athletix_Performance_Trends {


    public static function init() {


    }





    public static function calculate(
        $entity_id,
        $metric
    ) {


        $history =
            get_post_meta(

                $entity_id,

                '_athletix_history',

                true

            );



        if(
            ! is_array(
                $history
            )
        ) {

            return [];

        }



        $values = [];



        foreach(
            $history as $item
        ) {


            if(
                isset(
                    $item[$metric]
                )
            ) {


                $values[] =
                    $item[$metric];


            }


        }



        return $values;


    }


}

/**
 * Athletix Leaderboards
 */
class Athletix_Leaderboards {


    public static function init() {


    }





    public static function top_players(
        $metric,
        $limit = 10
    ) {


        $players =
            get_posts(

                [

                    'post_type'=>
                        'ax_player',


                    'posts_per_page'=>
                        $limit,


                    'meta_key'=>
                        '_athletix_' .
                        sanitize_key(
                            $metric
                        ),


                    'orderby'=>
                        'meta_value_num',


                    'order'=>
                        'DESC'

                ]

            );



        return $players;


    }


}

/**
 * Athletix Report Export
 */
class Athletix_Export_Reports {


    public static function init() {


        add_action(

            'admin_post_athletix_export_report',

            [

                __CLASS__,

                'export'

            ]

        );


    }





    public static function export() {


        if(
            ! current_user_can(
                'manage_athletix'
            )
        ) {

            return;

        }



        header(

            'Content-Type:text/csv'

        );


        header(

            'Content-Disposition:attachment; filename=athletix-report.csv'

        );



        echo "Athletix Report\n";

        echo "Generated," .
            current_time(
                'mysql'
            );



        exit;


    }


}
