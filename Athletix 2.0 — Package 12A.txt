/**
 * Athletix Roles
 */
class Athletix_Roles {


    public static function init() {


        add_action(

            'init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        add_role(

            'athletix_manager',

            'Athletix League Manager',

            [

                'read'=>true,

                'manage_athletix'=>true,

                'edit_athletix'=>true,

                'publish_athletix'=>true

            ]

        );



        add_role(

            'athletix_coach',

            'Athletix Coach',

            [

                'read'=>true,

                'view_players'=>true,

                'submit_statistics'=>true

            ]

        );



        add_role(

            'athletix_statistician',

            'Athletix Statistician',

            [

                'read'=>true,

                'edit_statistics'=>true,

                'publish_statistics'=>true

            ]

        );


    }


}

/**
 * Athletix Capabilities
 */
class Athletix_Capabilities {


    public static function init() {


        add_action(

            'admin_init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        $roles = [

            'administrator',

            'athletix_manager'

        ];



        foreach(
            $roles as $role_name
        ) {


            $role =
                get_role(
                    $role_name
                );



            if(
                $role
            ) {


                $role->add_cap(

                    'manage_athletix'

                );


                $role->add_cap(

                    'edit_athletix'

                );


            }


        }


    }


}

/**
 * Athletix Permission Checker
 */
class Athletix_Permissions {


    public static function init() {


    }





    public static function can_manage() {


        return current_user_can(

            'manage_athletix'

        );


    }





    public static function can_edit_stats() {


        return current_user_can(

            'edit_statistics'

        );


    }





    public static function can_view_players() {


        return current_user_can(

            'view_players'

        );


    }


}
