/**
 * Athletix Media Manager
 */
class Athletix_Media_Manager {


    public static function init() {


    }





    public static function attach(
        $post_id,
        $media_id
    ) {


        add_post_meta(

            $post_id,

            '_athletix_media',

            absint(

                $media_id

            )

        );


    }





    public static function get(
        $post_id
    ) {


        return get_post_meta(

            $post_id,

            '_athletix_media'

        );


    }


}

/**
 * Athletix Gallery System
 */
class Athletix_Gallery {


    public static function init() {


        add_shortcode(

            'athletix_gallery',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render(
        $atts
    ) {


        $post_id =
            $atts['id']
            ??
            get_the_ID();



        $media =
            Athletix_Media_Manager::get(

                $post_id

            );



        if(
            empty(
                $media
            )
        ) {

            return 'No media available.';

        }



        ob_start();


        ?>


        <div class="athletix-gallery">


        <?php foreach(
            $media as $image
        ) : ?>


            <div class="gallery-item">


                <?php echo wp_get_attachment_image(

                    $image,

                    'medium'

                ); ?>


            </div>


        <?php endforeach; ?>


        </div>


        <?php


        return ob_get_clean();


    }


}

/**
 * Athletix Content Feed
 */
class Athletix_Content_Feed {


    public static function init() {


        add_shortcode(

            'athletix_feed',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        $posts =
            get_posts(

                [

                    'post_type'=>
                        [

                            'post',

                            'ax_team'

                        ],


                    'posts_per_page'=>
                        10

                ]

            );



        ob_start();


        foreach(
            $posts as $post
        ) {


            echo '<div class="athletix-feed-item">';


            echo esc_html(

                $post->post_title

            );


            echo '</div>';


        }



        return ob_get_clean();


    }


}
