<?php

if ( ! defined( 'ABSPATH' ) ) {
    exit;
}


/**
 * Athletix Service Registry
 *
 * Lightweight dependency container.
 */
class Athletix_Registry {


    private static $items = [];




    /**
     * Initialize services.
     */
    public static function init() {


        self::register_defaults();


    }





    /**
     * Register default services.
     */
    private static function register_defaults() {


        self::register(

            'team.repository',

            new Athletix_Team_Repository()

        );


        self::register(

            'player.repository',

            new Athletix_Player_Repository()

        );


        self::register(

            'match.repository',

            new Athletix_Match_Repository()

        );


        self::register(

            'league.repository',

            new Athletix_League_Repository()

        );


        self::register(

            'season.repository',

            new Athletix_Season_Repository()

        );


    }





    /**
     * Add service.
     */
    public static function register(
        $name,
        $service
    ) {


        self::$items[$name] =
            $service;


    }





    /**
     * Retrieve service.
     */
    public static function get(
        $name
    ) {


        return
            self::$items[$name]
            ??
            null;


    }





    /**
     * Check service.
     */
    public static function has(
        $name
    ) {


        return isset(
            self::$items[$name]
        );


    }



}
