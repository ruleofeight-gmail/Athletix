/**
 * Match Repository
 *
 * Handles match data access.
 */
class Athletix_Match_Repository
extends Athletix_Base_Repository {


    protected $post_type =
        'ax_match';





    /**
     * Get home team.
     */
    public function home_team(
        $match
    ) {


        return Athletix_Match_Participants::get_home_team(
            $match
        );


    }





    /**
     * Get away team.
     */
    public function away_team(
        $match
    ) {


        return Athletix_Match_Participants::get_away_team(
            $match
        );


    }





    /**
     * Get match score.
     */
    public function score(
        $match
    ) {


        return $this->get_meta(

            $match,

            '_athletix_scores',

            [

                'home'=>0,
                'away'=>0

            ]

        );


    }





    /**
     * Get season.
     */
    public function season(
        $match
    ) {


        return $this->get_meta(

            $match,

            '_athletix_season',

            0

        );


    }


}

/**
 * League Repository
 */
class Athletix_League_Repository
extends Athletix_Base_Repository {


    protected $post_type =
        'ax_league';





    /**
     * Get seasons.
     */
    public function seasons(
        $league
    ) {


        return Athletix_Relationship_Engine::get(

            $league,

            'league_season'

        );


    }





    /**
     * Get league teams.
     */
    public function teams(
        $league
    ) {


        $seasons =
            $this->seasons(
                $league
            );



        $teams = [];



        foreach (
            $seasons as $season
        ) {


            $teams =
                array_merge(

                    $teams,

                    Athletix_Relationship_Engine::get(

                        $season,

                        'season_team'

                    )

                );


        }



        return array_unique(
            $teams
        );


    }


}
