/**
 * Athletix League Post Type

class Athletix_League_Post_Type {


    public static function init() {


        add_action(
            'init',
            [
                __CLASS__,
                'register'
            ]
        );


    }



    public static function register() {


        register_post_type(
            'ax_league',
            [

                'labels' => [

                    'name' =>
                        'Leagues',

                    'singular_name' =>
                        'League'

                ],


                'public' =>
                    true,


                'show_in_rest' =>
                    true,


                'menu_icon' =>
                    'dashicons-awards',


                'supports' => [

                    'title',
                    'thumbnail',
                    'editor'

                ],


                'rewrite' => [

                    'slug' =>
                        'leagues'

                ]

            ]
        );


    }


}

Now update class-plugin.php

We need to tell the framework that these modules exist.

Replace the load_core() method section with this expanded version.

File:

includes/class-plugin.php

Replace:

private static function load_core()

with:

private static function load_core() {


    $modules = [

        /*
         * Core Framework
         */
        'Athletix_Hooks',
        'Athletix_Events',
        'Athletix_Config',
        'Athletix_Registry',
        'Athletix_Cache',
        'Athletix_Logger',
        'Athletix_Validator',


        /*
         * Post Types
         */
        'Athletix_League_Post_Type',
        'Athletix_Season_Post_Type',
        'Athletix_Team_Post_Type',
        'Athletix_Player_Post_Type',
        'Athletix_Match_Post_Type',

    ];



    foreach (
        $modules as $module
    ) {


        if (
            class_exists(
                $module
            )
        ) {


            if (
                method_exists(
                    $module,
                    'init'
                )
            ) {


                $module::init();


            }


        }


    }


}
