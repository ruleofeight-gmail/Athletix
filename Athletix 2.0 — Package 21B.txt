/**
 * Athletix Payment History
 */
class Athletix_Payment_History {


    public static function init() {


    }





    public static function add(
        $user_id,
        $amount,
        $status = 'completed'
    ) {


        $history =
            get_user_meta(

                $user_id,

                '_athletix_payment_history',

                true

            );



        if(
            ! is_array(
                $history
            )
        ) {

            $history = [];

        }



        $history[] = [

            'amount'=>

                floatval(

                    $amount

                ),


            'status'=>

                sanitize_key(

                    $status

                ),


            'date'=>

                current_time(

                    'mysql'

                )

        ];



        update_user_meta(

            $user_id,

            '_athletix_payment_history',

            $history

        );



        return true;


    }





    public static function get(
        $user_id
    ) {


        return get_user_meta(

            $user_id,

            '_athletix_payment_history',

            true

        );


    }


}

/**
 * Athletix Refund System
 */
class Athletix_Refunds {


    public static function init() {


    }





    public static function create(
        $payment_id,
        $amount,
        $reason = ''
    ) {


        $refund_id =
            wp_insert_post(

                [

                    'post_type'=>
                        'ax_refund',


                    'post_title'=>
                        'Refund - ' .
                        $payment_id,


                    'post_status'=>
                        'publish'

                ]

            );



        update_post_meta(

            $refund_id,

            '_athletix_payment',

            absint(

                $payment_id

            )

        );



        update_post_meta(

            $refund_id,

            '_athletix_amount',

            floatval(

                $amount

            )

        );



        update_post_meta(

            $refund_id,

            '_athletix_reason',

            sanitize_text_field(

                $reason

            )

        );



        return $refund_id;


    }


}

/**
 * Athletix Financial Reports
 */
class Athletix_Financial_Reports {


    public static function init() {


    }





    public static function revenue(
        $users = []
    ) {


        $total = 0;



        foreach(
            $users as $user_id
        ) {


            $payments =
                Athletix_Payment_History::get(

                    $user_id

                );



            foreach(
                (array)$payments as $payment
            ) {


                if(
                    $payment['status']
                    ===
                    'completed'
                ) {


                    $total +=
                        floatval(

                            $payment['amount']

                        );


                }


            }


        }



        return $total;


    }


}
