<?php

if ( ! defined( 'ABSPATH' ) ) {
    exit;

/**
 * Athletix Admin Tools
 */
class Athletix_Admin_Tools {


    public static function init() {


        add_action(

            'admin_menu',

            [

                __CLASS__,

                'register'

            ]

        );


        add_action(

            'admin_post_athletix_import',

            [

                __CLASS__,

                'import'

            ]

        );


    }





    public static function register() {


        add_submenu_page(

            'athletix',

            'Tools',

            'Tools',

            'manage_options',

            'athletix-tools',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        ?>


        <div class="wrap">


            <h1>
                Athletix Tools
            </h1>



            <h2>
                Import CSV
            </h2>



            <form method="post"
                enctype="multipart/form-data"
                action="<?php echo esc_url(
                    admin_url(
                        'admin-post.php'
                    )
                ); ?>">



                <input type="hidden"
                    name="action"
                    value="athletix_import"
                >



                <?php

                wp_nonce_field(
                    'athletix_import'
                );

                ?>



                <p>


                    <select name="type">


                        <option value="teams">

                            Teams

                        </option>


                        <option value="players">

                            Players

                        </option>


                        <option value="matches">

                            Matches

                        </option>


                    </select>


                </p>




                <p>

                    <input type="file"
                        name="file"
                    >

                </p>



                <?php

                submit_button(
                    'Import'
                );

                ?>


            </form>



        </div>


        <?php


    }





    public static function import() {


        check_admin_referer(
            'athletix_import'
        );



        if(
            empty(
                $_FILES['file']
            )
        ) {

            wp_die(
                'No file uploaded'
            );

        }



        $type =
            sanitize_text_field(

                $_POST['type']

            );



        $file =
            $_FILES['file']['tmp_name'];



        Athletix_Import_Manager::run(

            $type,

            $file

        );



        wp_safe_redirect(

            admin_url(
                'admin.php?page=athletix-tools'
            )

        );


        exit;


    }


}
