/**
 * Athletix Team Registration
 */
class Athletix_Team_Registration {


    public static function init() {


    }





    public static function register(
        $user_id,
        $team_data
    ) {


        $team_id =
            wp_insert_post(

                [

                    'post_type'=>
                        'ax_team',


                    'post_title'=>
                        sanitize_text_field(

                            $team_data['name']

                        ),


                    'post_status'=>
                        'publish'

                ]

            );



        if(
            $team_id
        ) {


            update_post_meta(

                $team_id,

                '_athletix_owner',

                absint(

                    $user_id

                )

            );


        }



        return $team_id;


    }


}

class Athletix_Membership_Plans {


    public static function init() {


        add_action(

            'init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_post_type(

            'ax_plan',

            [

                'labels'=>[

                    'name'=>
                        'Membership Plans',


                    'singular_name'=>
                        'Membership Plan'

                ],


                'public'=>
                    false,


                'show_ui'=>
                    true,


                'supports'=>[

                    'title',

                    'editor'

                ]

            ]

        );


    }





    public static function assign(
        $user_id,
        $plan_id
    ) {


        update_user_meta(

            $user_id,

            '_athletix_plan',

            absint(

                $plan_id

            )

        );


    }


}

/**
 * Athletix Eligibility Engine
 */
class Athletix_Eligibility {


    public static function init() {


    }





    public static function check(
        $player_id,
        $rules = []
    ) {


        foreach(
            $rules as $key=>$value
        ) {


            $player_value =
                get_post_meta(

                    $player_id,

                    '_athletix_' .
                    sanitize_key(

                        $key

                    ),

                    true

                );



            if(
                $player_value !=
                $value
            ) {

                return false;

            }


        }



        return true;


    }


}
