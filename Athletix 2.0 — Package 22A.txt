/**
 * Athletix REST API
 */
class Athletix_REST_API {


    public static function init() {


        add_action(

            'rest_api_init',

            [

                __CLASS__,

                'register_routes'

            ]

        );


    }





    public static function register_routes() {


        register_rest_route(

            'athletix/v1',

            '/teams',

            [

                'methods'=>
                    'GET',


                'callback'=>

                    [

                        __CLASS__,

                        'teams'

                    ]

            ]

        );



        register_rest_route(

            'athletix/v1',

            '/players',

            [

                'methods'=>
                    'GET',


                'callback'=>

                    [

                        __CLASS__,

                        'players'

                    ]

            ]

        );


    }





    public static function teams() {


        return get_posts(

            [

                'post_type'=>
                    'ax_team',


                'posts_per_page'=>
                    -1

            ]

        );


    }





    public static function players() {


        return get_posts(

            [

                'post_type'=>
                    'ax_player',


                'posts_per_page'=>
                    -1

            ]

        );


    }


}

/**
 * Athletix API Authentication
 */
class Athletix_API_Auth {


    public static function init() {


    }





    public static function verify(
        $request
    ) {


        $token =
            $request->get_header(

                'athletix-token'

            );



        if(
            empty(
                $token
            )
        ) {

            return false;

        }



        return true;


    }


}

/**
 * Athletix Webhooks
 */
class Athletix_Webhooks {


    public static function init() {


        add_action(

            'rest_api_init',

            [

                __CLASS__,

                'register'

            ]

        );


    }





    public static function register() {


        register_rest_route(

            'athletix/v1',

            '/webhook',

            [

                'methods'=>
                    'POST',


                'callback'=>

                    [

                        __CLASS__,

                        'receive'

                    ]

            ]

        );


    }





    public static function receive(
        $request
    ) {


        $data =
            $request->get_json_params();



        do_action(

            'athletix_webhook_received',

            $data

        );



        return [

            'received'=>true

        ];


    }


}
