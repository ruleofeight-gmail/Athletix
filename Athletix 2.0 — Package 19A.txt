/**
 * Athletix Analytics Engine
 */
class Athletix_Analytics_Engine {


    public static function init() {


    }





    public static function calculate(
        $type,
        $id
    ) {


        switch(
            $type
        ) {


            case 'team':

                return Athletix_Team_Analytics::get(

                    $id

                );



            case 'player':

                return Athletix_Player_Analytics::get(

                    $id

                );


        }



        return [];


    }


}

/**
 * Athletix Team Analytics
 */
class Athletix_Team_Analytics {


    public static function init() {


    }





    public static function get(
        $team_id
    ) {


        return [

            'wins'=>

                self::metric(

                    $team_id,

                    'wins'

                ),


            'losses'=>

                self::metric(

                    $team_id,

                    'losses'

                ),


            'points'=>

                self::metric(

                    $team_id,

                    'points'

                )

        ];


    }





    private static function metric(
        $id,
        $key
    ) {


        return absint(

            get_post_meta(

                $id,

                '_athletix_' . $key,

                true

            )

        );


    }


}

/**
 * Athletix Report Generator
 */
class Athletix_Report_Generator {


    public static function init() {


    }





    public static function create(
        $title,
        $data
    ) {


        $report_id =
            wp_insert_post(

                [

                    'post_type'=>
                        'ax_report',


                    'post_title'=>
                        sanitize_text_field(

                            $title

                        ),


                    'post_status'=>
                        'publish'

                ]

            );



        if(
            $report_id
        ) {


            update_post_meta(

                $report_id,

                '_athletix_report_data',

                $data

            );


        }



        return $report_id;


    }


}
