/**
 * Athletix Admin Menu
 */
class Athletix_Admin_Menu {


    public static function init() {


        add_action(
            'admin_menu',
            [
                __CLASS__,
                'register'
            ]
        );


    }





    public static function register() {


        add_menu_page(

            'Athletix',

            'Athletix',

            'manage_options',

            'athletix',

            [

                'Athletix_Admin_Dashboard',

                'render'

            ],

            'dashicons-awards',

            25

        );




        add_submenu_page(

            'athletix',

            'Settings',

            'Settings',

            'manage_options',

            'athletix-settings',

            [

                'Athletix_Admin_Settings',

                'render'

            ]

        );


    }



}

/**
 * Athletix Settings Page
 */
class Athletix_Admin_Settings {


    public static function init() {


        add_action(
            'admin_init',
            [
                __CLASS__,
                'register'
            ]
        );


    }





    public static function register() {


        register_setting(

            'athletix_settings',

            'ax_active_sport'

        );


    }





    public static function render() {


        ?>

        <div class="wrap">

            <h1>
                Athletix Settings
            </h1>


            <form method="post"
                action="options.php">


                <?php

                settings_fields(
                    'athletix_settings'
                );


                ?>


                <table class="form-table">


                    <tr>

                        <th>
                            Active Sport
                        </th>


                        <td>


                            <select name="ax_active_sport">


                                <option value="soccer">

                                    Soccer

                                </option>


                                <option value="basketball">

                                    Basketball

                                </option>


                            </select>


                        </td>

                    </tr>


                </table>


                <?php

                submit_button();

                ?>


            </form>


        </div>


        <?php


    }


}
