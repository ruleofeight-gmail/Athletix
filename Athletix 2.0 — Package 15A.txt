        wp_add_dashboard_widget(

            'athletix_overview',

            'Athletix Overview',

            [

                __CLASS__,

                'render'

            ]

        );


    }





    public static function render() {


        echo Athletix_Dashboard_Widgets::overview();


    }


}

/**
 * Dashboard Statistics
 */
class Athletix_Dashboard_Stats {


    public static function init() {


    }





    public static function get() {


        return [

            'teams'=>

                self::count(
                    'ax_team'
                ),


            'players'=>

                self::count(
                    'ax_player'
                ),


            'matches'=>

                self::count(
                    'ax_match'
                )

        ];


    }





    private static function count(
        $type
    ) {


        return wp_count_posts(

            $type

        )->publish;


    }


}

/**
 * Dashboard Notices
 */
class Athletix_Dashboard_Notices {


    public static function init() {


        add_action(

            'admin_notices',

            [

                __CLASS__,

                'show'

            ]

        );


    }





    public static function show() {


        if(
            ! current_user_can(
                'manage_athletix'
            )
        ) {

            return;

        }



        $teams =
            wp_count_posts(
                'ax_team'
            )->publish;



        if(
            ! $teams
        ) {


            echo '<div class="notice notice-warning">';

            echo '<p>';

            echo 'Athletix: Add your first team to begin setup.';

            echo '</p>';

            echo '</div>';


        }


    }


}
